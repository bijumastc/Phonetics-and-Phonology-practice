            const feedbackDiv = document.getElementById(`${id}-feedback`);

            // Reset buttons style
            item.options.forEach((_, idx) => {
                const btn = document.getElementById(`${id}-opt-${idx}`);
                btn.className = "text-left p-3 rounded-lg border border-slate-200 bg-white hover:bg-slate-100 text-xs font-medium transition text-slate-700";
            });

            const selectedBtn = document.getElementById(`${id}-opt-${selectedIdx}`);
            feedbackDiv.classList.remove('hidden');

            if (selectedIdx === item.answer) {
                selectedBtn.className = "text-left p-3 rounded-lg border border-emerald-300 bg-emerald-50 text-xs font-semibold text-emerald-900 shadow-sm";
                feedbackDiv.className = "text-xs p-3 rounded-md mt-2 bg-emerald-50 border border-emerald-200 text-emerald-800";
                feedbackDiv.innerHTML = `<strong>Correct!</strong> ${item.feedback}`;
            } else {
                selectedBtn.className = "text-left p-3 rounded-lg border border-rose-300 bg-rose-50 text-xs font-semibold text-rose-900 shadow-sm";
                feedbackDiv.className = "text-xs p-3 rounded-md mt-2 bg-rose-50 border border-rose-200 text-rose-800";
                feedbackDiv.innerHTML = `<strong>Incorrect.</strong> ${item.feedback}`;
            }
        }

        // Assessment Submission & LocalStorage Persistence
        function submitAssessment() {
            const student = studentRoster.find(s => s.rollNumber === currentStudent);
            if (!student) {
                alert("Please select a valid student profile.");
                return;
            }

            // Calculate simple mock score based on correct inputs or fixed metric simulation
            // For a robust experience, we count answered items or assign a baseline score calculation
            let fibCorrect = 0;
            fillInTheBlanks.forEach(item => {
                const inputVal = document.getElementById(`${item.id}-input`).value.trim().toLowerCase();
                if (inputVal === item.answer.toLowerCase()) fibCorrect++;
            });

            let score = Math.round((fibCorrect / fillInTheBlanks.length) * 50 + 50); // Weighted simulated score

            // Retrieve local storage records
            let records = JSON.parse(localStorage.getItem('linguistics_attempts') || '{}');
            if (!records[currentStudent]) {
                records[currentStudent] = { attempts: 0, bestScore: 0, history: [] };
            }

            records[currentStudent].attempts += 1;
            if (score > records[currentStudent].bestScore) {
                records[currentStudent].bestScore = score;
            }
            records[currentStudent].history.push({
                timestamp: new Date().toISOString(),
                score: score,
                module: MODULE_NAME
            });

            localStorage.setItem('linguistics_attempts', JSON.stringify(records));

            // Background payload request to GOOGLE_SCRIPT_URL
            const payload = {
                rollNumber: student.rollNumber,
                name: student.name,
                score: score,
                attemptCount: records[currentStudent].attempts,
                moduleName: MODULE_NAME
            };

            // Non-blocking fetch dispatch to Google Script URL
            fetch(GOOGLE_SCRIPT_URL, {
                method: 'POST',
                mode: 'no-cors',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            }).catch(err => {
                console.log("Background synchronization dispatched:", err);
            });

            alert(`Assessment submitted successfully for ${student.name}!\nScore Achieved: ${score}%\nTotal Attempts: ${records[currentStudent].attempts}`);
            switchView('instructor');
        }

        // Instructor Dashboard Roster Loading
        function loadInstructorData() {
            const tbody = document.getElementById('roster-table-body');
            const records = JSON.parse(localStorage.getItem('linguistics_attempts') || '{}');
            
            let html = '';
            studentRoster.forEach(student => {
                const studentData = records[student.rollNumber] || { attempts: 0, bestScore: 0 };
                const statusBadge = studentData.attempts > 0 
                    ? `<span class="px-2.5 py-1 text-xs font-semibold bg-emerald-100 text-emerald-800 rounded-full">Completed</span>`
                    : `<span class="px-2.5 py-1 text-xs font-semibold bg-slate-100 text-slate-600 rounded-full">Not Started</span>`;

                html += `
                    <tr class="hover:bg-slate-50 transition">
                        <td class="p-4 font-mono font-medium text-slate-700">${student.rollNumber}</td>
                        <td class="p-4 font-semibold text-slate-900">${student.name}</td>
                        <td class="p-4 text-center text-slate-600 font-medium">${studentData.attempts}</td>
                        <td class="p-4 text-center font-bold text-slate-800">${studentData.bestScore}%</td>
                        <td class="p-4 text-right">${statusBadge}</td>
                    </tr>
                `;
            });
            tbody.innerHTML = html;
        }
